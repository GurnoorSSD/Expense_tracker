<script>
let expenses = JSON.parse(localStorage.getItem("expenses")) || [];
let selectedIndex = null;

const categories = ["Food","Transport","Rent","Utilities","Entertainment","Shopping","Health","Education","Travel","Other"];

function saveData() {
    localStorage.setItem("expenses", JSON.stringify(expenses));
}

function addExpense() {
    let category = category.value.trim();
    let amount = amountInput.value.trim();
    let note = noteInput.value.trim();

    if (!category || !amount) {
        alert("Category and Amount are required");
        return;
    }

    let expense = {
        date: new Date().toISOString().split("T")[0],
        category,
        amount: parseInt(amount),
        note
    };

    expenses.push(expense);
    saveData();
    clearInputs();
    loadExpenses();
}

function loadExpenses() {
    expenseTable.innerHTML = "";
    let total = 0;
    let summary = {};

    expenses.forEach((exp, index) => {
        let row = document.createElement("tr");
        row.onclick = () => selectRow(index, row);

        row.innerHTML = `
            <td>${exp.date}</td>
            <td>${exp.category}</td>
            <td>${exp.amount}</td>
            <td>${exp.note}</td>
        `;
        expenseTable.appendChild(row);

        total += exp.amount;
        summary[exp.category] = (summary[exp.category] || 0) + exp.amount;
    });

    totalText.innerText = `Total Spent: ₹${total}`;

    let sumText = "";
    for (let cat in summary) {
        sumText += `${cat}: ₹${summary[cat]}\n`;
    }
    summaryText.innerText = sumText;
}

function selectRow(index, row) {
    document.querySelectorAll("tr").forEach(r => r.classList.remove("selected"));
    row.classList.add("selected");
    selectedIndex = index;
}

function loadSelected() {
    if (selectedIndex === null) {
        alert("Select a row first");
        return;
    }
    let exp = expenses[selectedIndex];
    category.value = exp.category;
    amountInput.value = exp.amount;
    noteInput.value = exp.note;
}

function updateExpense() {
    if (selectedIndex === null) {
        alert("Select an expense to update");
        return;
    }

    expenses[selectedIndex].category = category.value;
    expenses[selectedIndex].amount = parseInt(amountInput.value);
    expenses[selectedIndex].note = noteInput.value;

    saveData();
    clearInputs();
    loadExpenses();
}

function deleteExpense() {
    if (selectedIndex === null) {
        alert("Select an expense to delete");
        return;
    }

    if (!confirm("Are you sure?")) return;

    expenses.splice(selectedIndex, 1);
    selectedIndex = null;
    saveData();
    loadExpenses();
}

function clearInputs() {
    category.value = "";
    amountInput.value = "";
    noteInput.value = "";
}

function createCategoryButtons() {
    categories.forEach(cat => {
        let btn = document.createElement("button");
        btn.innerText = cat;
        btn.onclick = () => category.value = cat;
        categoryButtons.appendChild(btn);
    });
}

/* Element shortcuts */
const category = document.getElementById("category");
const amountInput = document.getElementById("amount");
const noteInput = document.getElementById("note");
const expenseTable = document.getElementById("expenseTable");
const totalText = document.getElementById("total");
const summaryText = document.getElementById("summary");
const categoryButtons = document.getElementById("categoryButtons");

createCategoryButtons();
loadExpenses();
</script>
